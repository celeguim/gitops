# argocd install
kubectl create namespace argocd
kubectl apply -n argocd --server-side --force-conflicts -f https://raw.githubusercontent.com/argoproj/argo-cd/stable/manifests/install.yaml

kubectl port-forward -n argocd service/argocd-server 8080:80
argocd admin initial-password -n argocd
kubectl -n argocd get secret argocd-initial-admin-secret -o jsonpath="{.data.password}" | base64 -d; echo

argocd login localhost:8080 --insecure


helm lint charts/microservice

helm template jvminfo-v10 charts/microservice \
  -f charts/microservice/examples/springboot-v10.yaml

helm upgrade --install jvminfo-v10 charts/microservice \
  -f charts/microservice/examples/springboot-v10.yaml

helm upgrade --install jvminfo-v8 charts/microservice \
  -f charts/microservice/examples/springboot-v8.yaml

helm install jvminfo-v9 charts/microservice \
  -f charts/microservice/examples/springboot-v9.yaml

helm upgrade --install jvminfo-v9 charts/microservice \
  -f charts/microservice/examples/springboot-v9.yaml
